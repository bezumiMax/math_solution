h\gamma'(t_0) + o(h) = \gamma(t_0+h) - \gamma(t_0) = \gamma(t_0+h)

\[
F(\gamma(t_0+h)) = F(h(\gamma'(t_0) + o(1))) = h F(\gamma'(t_0) + o(1))

\[
\frac{F(\gamma(t_0+h)) - F(\gamma(t_0))}{h} = \frac{F(\gamma(t_0+h))}{h} = F(\gamma'(t_0) + o(1)) \to F(\gamma'(t_0))

\[
\Rightarrow (F(\gamma(t_0)))' = F(\gamma'(t_0))
\]
\[
(F(\gamma(t)))' = \nabla F(\gamma(t)) \cdot \gamma'(t)
\]
\[
\nabla F(\gamma(t_0+h)) = \nabla F\left(\frac{\gamma(t_0+h)}{|\gamma(t_0+h)|}\right) = \nabla F\left(\frac{h\gamma'(t_0) + o(h)}{|h\gamma'(t_0) + o(h)|}\right) = \nabla F\left(\frac{\gamma'(t_0) + o(1)}{|\gamma'(t_0) + o(1)|}\right)

|\nabla F(\gamma(t))| \le M \Rightarrow |(F(\gamma(t)))'| \le M |\gamma'(t)| \to 0 = \gamma'(t_0),

\noindent \textbf{N 4}

\[
\|A\| := \sup_{\|x\| \le 1} \|Ax\|
\]

\[
(Ax)_i = \sum_{j=1}^n a_{ij} x_j
\]

\[
\|Ax\|^2 = \sum_{i=1}^m \left( \sum_{j=1}^n a_{ij} x_j \right)^2
\]

\[
\|Ax\|^2 \le \sum_{i=1}^m \left( \sum_{j=1}^n a_{ij}^2 \right) \|x\|^2 = \|x\|^2 \left( \sum_{i=1}^m \sum_{j=1}^n a_{ij}^2 \right)
\]

\[
\text{при } \|x\| \le 1: \quad \|Ax\| \le \left( \sum a_{ij}^2 \right)^{\frac{1}{2}}
\]

\[
\Downarrow
\]

\[
\|A\| \le \left( \sum a_{ij}^2 \right)^{\frac{1}{2}}
\]

\noindent \textbf{N 5}

\[
y_0 \in I ; \quad \Phi(y) = \int_E f(x,y) dx
\]

\[
(!): \Phi(y) \text{ --- дифф-ма в т. } y_0 \text{ и } \frac{d\Phi(y)}{dy} = \int_E \frac{\partial f(x,y)}{\partial y} dx
\]

\[
h \neq 0, \quad y_0+h \in I
\]

\[
\frac{\Phi(y_0+h) - \Phi(y_0)}{h} = \int_E \frac{f(x, y_0+h) - f(x, y_0)}{h} dx
\]

\[
g_h(x) = \frac{f(x, y_0+h) - f(x, y_0)}{h} = \frac{\partial f}{\partial y}(x, \xi), \quad \xi \in (y_0, y_0+h)
\]

\[
\left| \frac{\partial f}{\partial y}(x, \xi) \right| \le F(x), \quad F(x) \text{ --- мажоранта}
\]

\[
\lim_{h \to 0} g_h(x) = \frac{\partial f}{\partial y}(x, y_0)
\]

\[
\text{по т. Лебега о мажорир. сх.:}
\]

\[
\lim_{h \to 0} \int_E g_h(x) dx = \int_E \lim_{h \to 0} g_h(x) dx
\]

\[
\Rightarrow \lim_{h \to 0} \int_E \frac{f(x, y_0+h) - f(x, y_0)}{h} dx = \int_E \frac{\partial f(x, y_0)}{\partial y} dx
\]

\[
\frac{d}{dy} \int_E f(x, y_0) dx = \int_E \frac{\partial f(x, y_0)}{\partial y} dx
\]
