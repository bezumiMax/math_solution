\section*{№1}

\[ |M|(E) = \sup \sum_{i=1}^{\infty} |M(E_i)| \]

1) Пусть \(E_i = \emptyset\) при \(i > N\):
\[ \sup \sum_{i=1}^{N} |M(E_i)| = \sup \sum_{i=1}^{\infty} |M(E_i)| \le |M|(E) \Rightarrow \]
\[ \Rightarrow \lambda(E) \le |M|(E) \]

2) \(\forall \varepsilon > 0 \ \exists \{A_i\}_{i=1}^{\infty} : \sum_{i=1}^{\infty} |M(A_i)| > |M|(E) - \frac{\varepsilon}{2}\)
\[ \sum_{i=1}^{\infty} |M(A_i)| - \text{сх.} \Rightarrow \exists N: \sum_{i=N+1}^{\infty} |M(A_i)| < \frac{\varepsilon}{2} \]
Пусть \(E_1 = A_1, \dots, E_N = A_N, E_{N+1} = \bigcup_{i=N+1}^{\infty} A_i\)
\[ \sum_{i=1}^{N+1} |M(E_i)| = \sum_{i=1}^{N} |M(A_i)| + |M(\bigcup_{i=N+1}^{\infty} A_i)| \]
\[ \sum_{i=1}^{N+1} |M(E_i)| \ge \sum_{i=1}^{N} |M(A_i)| = \sum_{i=1}^{\infty} |M(A_i)| - \sum_{i=N+1}^{\infty} |M(A_i)| > \]
\[ > (|M|(E) - \frac{\varepsilon}{2}) - \frac{\varepsilon}{2} = |M|(E) - \varepsilon \]
\[ \lambda(E) \ge \sum_{i=1}^{N+1} |M(E_i)| > |M|(E) - \varepsilon \]
\[ \lambda(E) \ge |M|(E) \]

\section*{№3}

\[ f \in L^p([0, 1]) \Leftrightarrow \int_0^1 |f(x)|^p dx < \infty \]
\[ f \in L^{\infty}([0, 1]) \Leftrightarrow \sup_{x \in [0, 1]} |f(x)| < \infty \text{ п.в.} \]
\[ \int_0^1 |f(x)|^p dx \le \int_0^1 M^p dx \le M^p \quad \forall p : 1 \le p < \infty \]
\[ \Rightarrow f \in L^p([0, 1]) \]
\[ f = \ln x \]
\[ \int_0^1 |\ln x|^p dx = \left| \begin{array}{l} x = e^{-t} \\ dx = -e^{-t} dt \\ x(0) = +\infty \\ x(1) = 0 \end{array} \right| = \int_0^{+\infty} t^p e^{-t} dt = \int_0^1 t^p e^{-t} dt + \int_1^{+\infty} t^p e^{-t} dt \]
\[ 0 \le t^p e^{-t} \le 1 \text{ при } t \in [0, 1] \]
\[ t^p \le e^{t/2} \text{ при } t \to +\infty \Rightarrow t^p e^{-t} \le e^{-t/2} \]
\[ \int_1^{\infty} t^p e^{-t} dt \le \int_1^{\infty} e^{-t/2} dt = 2e^{-1/2} < \infty \]
\[ \Rightarrow f \in \bigcup_{1 \le p < \infty} L^p([0, 1]), \text{ но } f \notin L^{\infty}([0, 1]) \]
\[ \Rightarrow L^{\infty}([0, 1]) \subset \bigcup_{1 \le p < \infty} L^p([0, 1]) \]

\[ \Rightarrow |\lambda|(E) = 0 \Rightarrow \text{если } \lambda \ll \mu, \text{ то } |\lambda| \ll \mu. \text{ Пусть } \lambda \ll \mu \]

Пусть \(\exists \varepsilon > 0 \ \forall \delta > 0 \ \exists E \in \mathcal{M}: \mu(E) < \delta, \ |\lambda|(E) \ge \varepsilon\)
\[ \{E_n\}_{n=1}^{\infty} : \delta = 2^{-n} \Rightarrow \mu(E_n) < 2^{-n}, \ |\lambda|(E_n) \ge \varepsilon \]
\[ \sum_{n=1}^{\infty} \mu(E_n) < \infty \Rightarrow \mu(\limsup_{n\to\infty} E_n) = 0 \Rightarrow |\lambda|(\limsup_{n\to\infty} E_n) = 0 \]
\[ |\lambda|(\bigcup_{k \ge n} E_k) \ge |\lambda|(E_n) \ge \varepsilon \]
\[ |\lambda|(\limsup_{n\to\infty} E_n) = |\lambda|(\bigcap_{n=1}^{\infty} \bigcup_{k \ge n} E_k) = \lim_{n\to\infty} |\lambda|(\bigcup_{k \ge n} E_k) \ge \varepsilon \]
\text{св-во непрерывности} \(\Rightarrow\) противоречие. \text{(Теорема Бореля-Кантелли)}
